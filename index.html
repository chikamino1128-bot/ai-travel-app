function mergeItineraryText(currentTrip, inputText) { const updatedTrip = JSON.parse(JSON.stringify(currentTrip)); const lines = inputText.split('\n'); let currentDayNum = 1; lines.forEach(line => { const trimmed = line.trim(); if (!trimmed) return; const dayMatch = trimmed.match(/(?:🗓️)?\s*(\d+)日目/); if (dayMatch) { currentDayNum = parseInt(dayMatch[1], 10); return; } let currentSchedule = updatedTrip.schedule.find(s => s.day === currentDayNum); if (!currentSchedule) { currentSchedule = { day: currentDayNum, events: [] }; updatedTrip.schedule.push(currentSchedule); } const timeMatch = trimmed.match(/^(\d{1,2}:\d{2})(?:[〜~-](\d{1,2}:\d{2}))?\s*(.+)$/); if (timeMatch) { const startTime = timeMatch[1]; const endTime = timeMatch[2] || ''; const newTitle = timeMatch[3].trim(); const existingEvent = currentSchedule.events.find(e => e.time === startTime); if (existingEvent) { existingEvent.title = newTitle; if (endTime) { existingEvent.endTime = endTime; } } else { const defaultMember = (updatedTrip.members && updatedTrip.members.length > 0) ? updatedTrip.members[0] : '自分'; currentSchedule.events.push({ id: 'evt_' + Date.now() + '_' + Math.random().toString(36).substr(2, 4), type: newTitle.includes('✈️') || newTitle.includes('🚗') ? 'move' : 'spot', title: newTitle, time: startTime, endTime: endTime, startDay: currentDayNum, endDay: currentDayNum, bookingStatus: 'unnecessary', expense: '', paidBy: defaultMember, forMembers: updatedTrip.members ? [...updatedTrip.members] : [], memo: '', photos: [] }); } } }); return updatedTrip; }
