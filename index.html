function mergeItineraryText(text) { const trip = getCurrentTrip(); if (!trip) return; const lines = text.split('\n'); let currentDayNum = 1; lines.forEach(line => { const trimmed = line.trim(); if (!trimmed) return; // 「1日目」「2日目」などの日付け指定を判定 const dayMatch = trimmed.match(/(?:🗓️)?\s*(\d+)日目/); if (dayMatch) { currentDayNum = parseInt(dayMatch[1]); return; } const currentSchedule = trip.schedule.find(s => s.day === currentDayNum); if (!currentSchedule) return; // 時間とタイトルの抽出（例: 13:15〜14:15 ランチ） const timeMatch = trimmed.match(/^(\d{1,2}:\d{2})(?:[〜~-](\d{1,2}:\d{2}))?\s*(.+)$/); if (timeMatch) { const startTime = timeMatch[1]; const endTime = timeMatch[2] || ''; const newTitle = timeMatch[3].trim(); // 同じ時間の既存イベントを検索 const existingEvent = currentSchedule.events.find(e => e.time === startTime); if (existingEvent) { // 【重要】既存データを壊さず、変更のあった項目（タイトル・終了時間）のみ上書き existingEvent.title = newTitle; if (endTime) { existingEvent.endTime = endTime; } // ※ expense, paidBy, memo, photos, bookingStatus などはノータッチで保持されます } else { // 新規作成時も、写真(photos)や費用(expense)などの全フィールドを初期値付きで用意 currentSchedule.events.push({ id: 'evt_' + Date.now() + Math.random().toString(36).substr(2, 4), type: newTitle.includes('✈️') || newTitle.includes('🚗') ? 'move' : 'spot', title: newTitle, time: startTime, endTime: endTime, startDay: currentDayNum, endDay: currentDayNum, bookingStatus: 'unnecessary', expense: '', paidBy: trip.members[0] || '自分', forMembers: [...trip.members], memo: '', photos: [] // 写真用配列も保持 }); } } }); saveTrips(); switchTab('itinerary'); }
